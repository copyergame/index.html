/** * IP Greeting Web Application * Single-file runnable server */ const http = require('http'); const PORT = process.env.PORT || 3000; // User IP mappings const USER_RULES = [ { name: 'Nasr', match: (ip) => ip === '94.204.74.75' }, { name: 'Hamza', match: (ip) => ip.startsWith('192.168.47') }, { name: 'Yassin', match: (ip) => ip === '92.99.64.6' } ]; const server = http.createServer((req, res) => { if (req.url === '/' || req.url === '') { // Extract client IP address (checks headers for proxies/cloud deployment) const xForwardedFor = req.headers['x-forwarded-for']; let rawIp = xForwardedFor ? xForwardedFor.split(',')[0] : req.socket.remoteAddress || ''; // Clean IPv6 wrapping (e.g., ::ffff:94.204.74.75 -> 94.204.74.75) and strip dynamic ports let clientIp = rawIp.replace(/^.*:/, '').trim(); // Check IP against rules const matchedUser = USER_RULES.find((rule) => rule.match(clientIp)); const greeting = matchedUser ? `hi ${matchedUser.name}` : 'hi Guest'; // Output single HTML page res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(` IP Greeting

${greeting}

Your IP: ${clientIp || 'Unknown'}

`); } else { res.writeHead(404, { 'Content-Type': 'text/plain' }); res.end('Not Found'); } }); server.listen(PORT, () => { console.log(`Server is live at http://localhost:${PORT}`); });
